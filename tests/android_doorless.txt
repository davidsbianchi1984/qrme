# Routes this shell cannot reach, with the reason. A ratchet: it may shrink
# and must not grow past the ceiling in test_the_phone_is_a_client_too.py.
#
# One row, and it says why. Every other route the backend serves is
# reachable from this shell.
#
#
# POST /rooms/{room_id}/heard — the route takes recorded audio and answers
# with words. This shell has no recorder wired to it yet (AVAudioRecorder /
# MediaRecorder / NAudio), and adding a binding with no caller would take the
# route off this backlog while building nothing — the exact failure
# unused_native_bindings.txt was written to catch. The web console has the
# door; these three do not, and this row says so until they do.
POST /rooms/{room_id}/heard

# POST /interactors/{interactor_id}/heard — the same door without a room, and
# this shell should never want it. It exists because the *browser's* own
# recogniser is ended when a page is put away, so the web console records and
# posts the bytes instead. This shell has no such problem: it holds a native
# recogniser that keeps working with the app off screen, does it on the
# device, and spends none of the deployment's transcription doing it.
#
#     asked     can this client reach the route
#     mattered  does this client have a better answer than the route
#
# Recorded rather than built, because building it would be giving a phone a
# slower, costlier ear than the one it already has — a door that exists to
# empty a backlog is worse than an honest row.
POST /interactors/{interactor_id}/heard

# The avatar registry (qrme/avatarreg.py) — the face ledger's doors. The
# console carries the whole card: the shelf, claiming, painting from
# words, stocking, retiring. The shells' portrait screens follow with
# their own round, the way the voice card's controls did — a rights
# surface ships first where its language lives.
GET /avatars/library
POST /avatars/library
GET /accounts/{account_id}/avatars
POST /accounts/{account_id}/avatars
POST /profiles/{profile_id}/avatar/claim
POST /profiles/{profile_id}/avatar/painted
DELETE /avatars/registry/{registry_id}
# GET /rooms/xr-platforms — the XR shelf is a discovery card the desktop
# Rooms screen draws; the shells ARE two of the platforms it lists, so a
# phone reading its own row adds nothing yet. Reachable when the native
# rooms screens grow the card.
GET /rooms/xr-platforms
# POST /avatars/library/pull — the operator's one-button shelf fill; an
# operator holds the signup key at a desk, not on a phone. Console-only,
# like the library stock door above it.
POST /avatars/library/pull

# POST /watch-parties/{party_id}/watch — have the party's video watched by
# the platform's own eyes and ears (qrme/watching.py). The web console
# holds this door on the party's context panel — the panel that shows
# what a profile in the room is told. This shell draws no context panel
# yet, and a watch button with no panel to land its viewing on would be
# a binding with no caller — the exact failure unused_native_bindings.txt
# was written to catch. The row leaves this backlog when the shell grows
# the panel.
POST /watch-parties/{party_id}/watch
